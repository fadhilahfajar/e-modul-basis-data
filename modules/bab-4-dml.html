DML — Memanipulasi Data

⏱ memuat... Menengah 5 soal quiz

Tujuan Pembelajaran

  • Menggunakan INSERT untuk menambahkan data ke tabel
  • Menggunakan SELECT dengan berbagai klausa: WHERE, ORDER BY, LIMIT, GROUP BY
  • Menerapkan fungsi agregat: COUNT, SUM, AVG, MAX, MIN
  • Menggunakan UPDATE dan DELETE dengan klausa WHERE yang tepat
  • Menjalankan query SQL langsung di browser melalui SQL Playground

INSERT — Mengisi Tabel dengan Data

SQL — INSERT 
-- Insert satu baris
INSERT INTO anggota (nis, nama_lengkap, kelas, email)
VALUES ('2024001', 'Andi Pratama', 'XI PPLG 1', 'andi@smk.sch.id');

-- Insert beberapa baris sekaligus (lebih efisien)
INSERT INTO buku (isbn, judul, pengarang, genre, stok, tahun_terbit) VALUES
  ('978-602-1', 'Pemrograman Web Dasar',    'Budi Santoso',  'Teknologi', 5, 2022),
  ('978-602-2', 'Basis Data Relasional',    'Citra Dewi',    'Teknologi', 3, 2023),
  ('978-602-3', 'Algoritma & Pemrograman',  'Dedi Wahyudi',  'Teknologi', 4, 2021),

  ('978-002-5', 'Laskar Pelangi',           'Andrea Hirata', 'Novel',     6, 2005);

SELECT — Membaca Data

SELECT adalah perintah yang paling sering digunakan. Strukturnya modular — tiap klausa bisa dikombinasikan sesuai kebutuhan.

SQL — SELECT dengan berbagai klausa 
-- SELECT dasar: tampilkan semua kolom semua baris
SELECT * FROM buku;

-- SELECT kolom tertentu dengan alias
SELECT isbn, judul AS 'Nama Buku', pengarang, stok AS 'Stok Tersedia'
FROM buku;

-- WHERE: filter baris berdasarkan kondisi
SELECT judul, stok
FROM buku
WHERE genre = 'Teknologi' AND stok > 2;

-- Operator LIKE: pencarian pola (% = banyak karakter, _ = satu karakter)
SELECT judul, pengarang
FROM buku
WHERE judul LIKE '%database%';  -- judul mengandung kata 'database'

-- BETWEEN: rentang nilai
SELECT judul, tahun_terbit
FROM buku
WHERE tahun_terbit BETWEEN 2021 AND 2023;

-- IN: sama dengan banyak OR
SELECT judul, genre
FROM buku
WHERE genre IN ('Teknologi', 'Sains');

-- IS NULL: cek nilai kosong
SELECT nama_lengkap FROM anggota WHERE email IS NULL;

-- ORDER BY: urutkan hasil (ASC = naik, DESC = turun)
SELECT judul, stok FROM buku
ORDER BY stok DESC, judul ASC;

-- LIMIT: batasi jumlah baris (berguna untuk paginasi)
SELECT judul FROM buku ORDER BY judul LIMIT 3;
SELECT judul FROM buku ORDER BY judul LIMIT 3 OFFSET 3; -- baris ke-4 dst

-- DISTINCT: hilangkan duplikat
SELECT DISTINCT genre FROM buku;

Fungsi Agregat & GROUP BY

SQL — Agregat & Grouping 
-- COUNT: hitung baris
SELECT COUNT(*) AS total_buku FROM buku;
SELECT COUNT(email) AS ada_email FROM anggota;  -- NULL tidak dihitung

-- SUM, AVG, MAX, MIN
SELECT
  SUM(stok)  AS total_stok,
  AVG(stok)  AS rata_stok,
  MAX(stok)  AS stok_terbanyak,
  MIN(stok)  AS stok_tersedikit
FROM buku;

-- GROUP BY: hitung agregat per kelompok
SELECT genre, COUNT(*) AS jml_buku, SUM(stok) AS total_stok
FROM buku
GROUP BY genre;

-- HAVING: filter setelah GROUP BY (bukan WHERE — berbeda!)
SELECT genre, COUNT(*) AS jml_buku
FROM buku
GROUP BY genre
HAVING COUNT(*) > 1;  -- hanya genre dengan lebih dari 1 buku

-- Urutan eksekusi SQL (penting!): FROM → WHERE → GROUP BY → HAVING → SELECT → ORDER BY → LIMIT
ℹ️
WHERE vs HAVING — Kapan Menggunakan Masing-masing?

WHERE memfilter baris sebelum dikelompokkan — tidak bisa menggunakan fungsi agregat. HAVING memfilter setelah GROUP BY — khusus untuk filter hasil agregat.

UPDATE & DELETE — Ubah dan Hapus Data

🚨
Aturan Wajib: Selalu Gunakan WHERE!

UPDATE dan DELETE tanpa WHERE akan mengubah/menghapus SEMUA baris. Ini adalah kesalahan fatal yang sangat sering terjadi. Kebiasaan baik: tulis WHERE dulu sebelum menulis SET atau perintah lainnya.

SQL — UPDATE & DELETE 
-- UPDATE: ubah data
-- Step 1: verifikasi dulu dengan SELECT
SELECT * FROM anggota WHERE id_anggota = 3;

-- Step 2: baru eksekusi UPDATE
UPDATE anggota
SET email = 'citra.baru@smk.sch.id', kelas = 'XII PPLG 1'
WHERE id_anggota = 3;

-- Update berdasarkan kondisi
UPDATE buku
SET stok = stok - 1
WHERE isbn = '978-602-2';

-- DELETE: hapus baris
-- Pastikan WHERE sudah benar sebelum eksekusi!
DELETE FROM peminjaman
WHERE status = 'dikembalikan' AND tgl_kembali < '2024-01-01';
